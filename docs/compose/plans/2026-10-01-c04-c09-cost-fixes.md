// src/gyccode/billing/reconciliation.ts
export function reconcileBilling(
  localCosts: Map<string, number>, // sessionID -> local cost
  providerCosts: Map<string, number>, // sessionID -> provider cost
  threshold = 0.01,
): ReconciliationReport {
  const allSessions = new Set([...localCosts.keys(), ...providerCosts.keys()])
  const rows = Array.from(allSessions).map((sessionID) => {
    const local = localCosts.get(sessionID) ?? 0
    const provider = providerCosts.get(sessionID) ?? 0
    const drift = provider - local
    const driftRatio = provider === 0 ? 0 : Math.abs(drift) / provider
    return { sessionID, local, provider, drift, driftRatio, mismatched: driftRatio > threshold }
  })
  return {
    generatedAt: new Date().toISOString(),
    totalSessions: rows.length,
    mismatchedCount: rows.filter((r) => r.mismatched).length,
    totalDrift: rows.reduce((sum, r) => sum + r.drift, 0),
    rows,
  }
}
```

// src/core/database/db.ts
export function promptCacheStats(sessionID: SessionSchema.ID): Effect.Effect<CacheStats> {
  return Effect.gen(function* () {
    const { db } = yield* Database.Service
    const rows = yield* db
      .selectFrom(SessionMessageTable)
      .select(["data", "time_created"])
      .where("session_id", "=", sessionID)
      .orderBy("seq", "asc")
      .execute()
      .pipe(Effect.orDie)

    // 复用 dialog-cost.tsx:86-105 的计算逻辑
    let hit = 0, base = 0, prevTotal = 0
    for (const row of rows) {
      const data = JSON.parse(row.data)
      if (data.role === "assistant" && data.tokens) {
        const inclusive = (data.tokens.input ?? 0) + (data.tokens.cache?.read ?? 0) + (data.tokens.cache?.write ?? 0)
        if (base > 0) {
          hit += Math.min(data.tokens.cache?.read ?? 0, prevTotal)
          base += prevTotal
        }
        prevTotal = inclusive
      }
    }
    return { prefixBase: base, prefixHit: hit, steadyBase: 0, steadyHit: 0 }
  })
}
```
