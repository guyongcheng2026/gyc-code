```typescript
// 1. 在 GlobalBus 中添加清理方法
export const GlobalBus = {
  on(event: string, handler: EventHandler): Disposable {
    const listeners = getListeners(event)
    listeners.add(handler)
    return {
      dispose: () => {
        listeners.delete(handler)
        // 记录 listener 计数用于调试
      }
    }
  },
  off(event: string, handler: EventHandler): void {
    getListeners(event).delete(handler)
  }
}

// 2. 在 worker 生命周期中正确清理
class WorkerManager {
  private disposables: Disposable[] = []
  
  initialize() {
    const disposable = GlobalBus.on("event", handler)
    this.disposables.push(disposable)
  }
  
  restart() {
    // 重启前清理所有监听器
    for (const d of this.disposables) {
      d.dispose()
    }
    this.disposables = []
  }
}
```

```typescript
    const now = Date.now()
    for (const [key, timestamp] of modCacheTimestamps.entries()) {
      if (now - timestamp > MOD_CACHE_TTL) {
        modCache.delete(key)
        modCacheTimestamps.delete(key)
      }
    }
  }, CLEANUP_INTERVAL)
}

function stopCleanup() {
  if (cleanupTimer) {
    clearInterval(cleanupTimer)
    cleanupTimer = null
  }
}
```
